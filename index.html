<script>

    (function() {

        const dimBtns = document.querySelectorAll('.dim-btn');

        const previewCanvas = document.getElementById('previewCanvas');

        let ctx = previewCanvas.getContext('2d');

        let currentWidth = 640, currentHeight = 360;

        let animFrame = null;

        let frameIdx = 0;

        

        const dimMap = {

            '320x180': { w: 320, h: 180 },

            '640x360': { w: 640, h: 360 },

            '960x540': { w: 960, h: 540 }

        };

        

        function setDimension(dimStr) {

            const dim = dimMap[dimStr];

            if (!dim) return;

            currentWidth = dim.w;

            currentHeight = dim.h;

            previewCanvas.width = currentWidth;

            previewCanvas.height = currentHeight;

            if (animFrame) cancelAnimationFrame(animFrame);

            frameIdx = 0;

            animatePreview();

        }

        

        // አንድ ፍሬም መሳል (ለቅድመ እይታ እና ለ GIF)

        function drawBingoFrame(context, w, h, frameNumber, totalFrames = 45) {

            const progress = frameNumber / totalFrames;

            context.clearRect(0, 0, w, h);

            

            // ዳራ

            const grad = context.createLinearGradient(0, 0, w, h);

            grad.addColorStop(0, '#1a2a6c');

            grad.addColorStop(0.5, '#b21f1f');

            grad.addColorStop(1, '#fdbb2d');

            context.fillStyle = grad;

            context.fillRect(0, 0, w, h);

            

            // ርዕስ

            context.fillStyle = '#FFD700';

            context.font = `bold ${Math.max(18, Math.floor(w/18))}px 'Segoe UI'`;

            context.textAlign = 'center';

            context.fillText('✨ ቢንጎ ጨዋታ ✨', w/2, h * 0.15);

            context.font = `${Math.max(12, Math.floor(w/35))}px system-ui`;

            context.fillStyle = 'white';

            context.fillText('የቴሌግራም ሚኒ አፕ', w/2, h * 0.25);

            

            // ካርድ

            const cardW = Math.min(w * 0.7, 300);

            const cardH = cardW;

            const cardX = (w - cardW)/2;

            const cardY = h * 0.35;

            const cellW = cardW / 5;

            const cellH = cardH / 5;

            

            context.fillStyle = 'rgba(13, 71, 161, 0.85)';

            context.fillRect(cardX, cardY, cardW, cardH);

            context.strokeStyle = '#ff9800';

            context.lineWidth = 2;

            context.strokeRect(cardX, cardY, cardW, cardH);

            

            const letters = ['B', 'I', 'N', 'G', 'O'];

            context.fillStyle = '#FFD700';

            context.font = `bold ${Math.max(12, Math.floor(cellW*0.4))}px system-ui`;

            for (let i=0; i<5; i++) {

                context.fillText(letters[i], cardX + i*cellW + cellW/2, cardY + cellH*0.6);

            }

            

            // ቁጥሮች (ምሳሌ)

            const sample = [

                [ 5,18,33,49,72],

                [12,24,38,55,68],

                [ 3,29, 0,52,70],

                [ 8,17,44,58,63],

                [14,22,41,60,75]

            ];

            const totalCells = 24;

            const markedCount = Math.min(totalCells, Math.floor(progress * totalCells) + 1);

            

            for (let row=0; row<5; row++) {

                for (let col=0; col<5; col++) {

                    let num = sample[row][col];

                    let isFree = (row===2 && col===2);

                    let cx = cardX + col*cellW + cellW/2;

                    let cy = cardY + row*cellH + cellH/2;

                    

                    let cellIdx = row*5 + col;

                    let isMarked = false;

                    if(isFree) isMarked = true;

                    else {

                        let pseudo = (cellIdx * 131) % totalCells;

                        if(pseudo < markedCount) isMarked = true;

                    }

                    

                    if(isMarked && !isFree) {

                        context.fillStyle = '#4CAF50';

                        context.fillRect(cardX+col*cellW, cardY+row*cellH, cellW, cellH);

                        context.fillStyle = 'white';

                    } else if(isFree) {

                        context.fillStyle = '#FF9800';

                        context.fillRect(cardX+col*cellW, cardY+row*cellH, cellW, cellH);

                        context.fillStyle = '#000';

                    } else {

                        context.fillStyle = '#1976D2';

                        context.fillRect(cardX+col*cellW, cardY+row*cellH, cellW, cellH);

                        context.fillStyle = '#eee';

                    }

                    context.strokeStyle = '#aaa';

                    context.strokeRect(cardX+col*cellW, cardY+row*cellH, cellW, cellH);

                    

                    if(isFree) {

                        context.font = `${Math.max(10, Math.floor(cellW*0.2))}px system-ui`;

                        context.fillStyle = '#000';

                        context.fillText('ነፃ', cx, cy);

                    } else {

                        context.font = `${Math.max(12, Math.floor(cellW*0.35))}px monospace`;

                        context.fillStyle = '#fff';

                        context.fillText(num.toString(), cx, cy);

                    }

                }

            }

            

            // ተንሳፋፊ ኳስ

            const ballX = w * 0.2 + Math.sin(frameNumber * 0.2) * w * 0.1;

            const ballY = h * 0.75 + Math.cos(frameNumber * 0.15) * h * 0.05;

            context.beginPath();

            context.arc(ballX, ballY, h * 0.06, 0, Math.PI*2);

            context.fillStyle = '#ff5722';

            context.fill();

            context.fillStyle = 'white';

            context.font = `bold ${Math.max(12, Math.floor(h*0.045))}px monospace`;

            context.fillText('B12', ballX, ballY+2);

            

            // ደስታ (confetti) መጨረሻ ላይ

            if(progress > 0.7) {

                const intensity = (progress - 0.7)/0.3;

                for(let i=0; i<12*intensity; i++) {

                    const fx = (Math.sin(frameNumber*0.3 + i) * w*0.4) + w/2;

                    const fy = (Math.cos(frameNumber*0.5 + i) * h*0.3) + h/2;

                    context.fillStyle = `hsl(${Date.now() + i*36}, 80%, 60%)`;

                    context.fillRect(fx, fy, 4, 4);

                }

            }

            if(progress > 0.85) {

                context.font = `bold ${Math.max(22, Math.floor(w/12))}px system-ui`;

                context.fillStyle = '#FFD700';

                context.shadowBlur = 8;

                context.fillText('🎉 ቢንጎ! 🎉', w/2, h*0.92);

                context.shadowBlur = 0;

            }

        }

        

        function animatePreview() {

            drawBingoFrame(ctx, currentWidth, currentHeight, frameIdx % 45, 45);

            frameIdx++;

            animFrame = requestAnimationFrame(animatePreview);

        }

        

        // እውነተኛ GIF ማመንጨት

        function generateRealGIF() {

            const statusDiv = document.getElementById('statusMsg');

            statusDiv.innerHTML = '⏳ GIF እየተዘጋጀ ነው... እባክዎ ይጠብቁ (እስከ 10 ሰከንድ)';

            statusDiv.style.color = '#ffaa00';

            

            const w = currentWidth;

            const h = currentHeight;

            const totalFrames = 45;

            const delay = 33;

            

            const offCanvas = document.createElement('canvas');

            offCanvas.width = w;

            offCanvas.height = h;

            const offCtx = offCanvas.getContext('2d');

            

            const gif = new GIF({

                workers: 2,

                quality: 15,

                width: w,

                height: h,

                workerScript: 'https://cdn.jsdelivr.net/npm/gif.js@0.2.0/dist/gif.worker.js'

            });

            

            for(let i=0; i<totalFrames; i++) {

                drawBingoFrame(offCtx, w, h, i, totalFrames);

                gif.addFrame(offCanvas, { copy: true, delay: delay });

            }

            

            gif.on('finished', function(blob) {

                const url = URL.createObjectURL(blob);

                const a = document.createElement('a');

                a.href = url;

                a.download = `bingo_${w}x${h}.gif`;

                document.body.appendChild(a);

                a.click();

                document.body.removeChild(a);

                URL.revokeObjectURL(url);

                statusDiv.innerHTML = '✅ GIF ተወርዷል! አሁን ወደ BotFather ላኩት።';

                statusDiv.style.color = '#a5d6a7';

            });

            gif.on('error', function(e) {

                console.error(e);

                statusDiv.innerHTML = '❌ GIF መፍጠር አልተቻለም። እባክዎ ገጹን እንደገና ጫኑ ወይም ትንሽ መጠን ይሞክሩ።';

                statusDiv.style.color = '#ff8888';

            });

            gif.render();

        }

        

        // Event listeners

        dimBtns.forEach(btn => {

            btn.addEventListener('click', () => {

                dimBtns.forEach(b => b.classList.remove('active'));

                btn.classList.add('active');

                const dim = btn.getAttribute('data-dim');

                setDimension(dim);

            });

        });

        document.getElementById('downloadGifBtn').addEventListener('click', generateRealGIF);

        

        // መጀመሪያ

        setDimension('640x360');

    })();

</script>
